# CMakeLists.txt for the libb2 0.98.1 source tree, whose own build is autotools only: BuildLibB2.cmake
# copies this file over <source dir>/CMakeLists.txt after the tarball is extracted.
#
# Builds the portable C implementation (no SSE/AVX dispatch, "--disable-fat --disable-sse" in autotools
# terms): blake2s-ref.c, blake2b-ref.c, blake2sp.c, blake2bp.c with an empty SUFFIX. That is the variant that
# also builds for emscripten, wasm, musl and the 32-bit targets.
cmake_minimum_required(VERSION 3.10)
project(libb2 C)

include(CheckFunctionExists)

check_function_exists(explicit_bzero HAVE_EXPLICIT_BZERO)
check_function_exists(explicit_memset HAVE_EXPLICIT_MEMSET)
check_function_exists(memset_s HAVE_MEMSET_S)
check_function_exists(memset HAVE_MEMSET)

# src/blake2-impl.h includes "config.h": write the few symbols it reads
set(CONFIG_H "/* generated by libb2.CMakeLists.txt */\n")

foreach(SYMBOL HAVE_EXPLICIT_BZERO HAVE_EXPLICIT_MEMSET HAVE_MEMSET_S HAVE_MEMSET)
  if(${SYMBOL})
    string(APPEND CONFIG_H "#define ${SYMBOL} 1\n")
  endif()
endforeach()

# the byte order is the compiler's to say (a cross build cannot run a test program, and
# CMAKE_C_BYTE_ORDER is empty for some toolchains)
string(APPEND CONFIG_H "#if defined(__BYTE_ORDER__) && defined(__ORDER_LITTLE_ENDIAN__) && __BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__\n#define NATIVE_LITTLE_ENDIAN 1\n#endif\n")

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/config.h" "${CONFIG_H}")

add_library(b2 STATIC
  src/blake2b-ref.c
  src/blake2s-ref.c
  src/blake2bp.c
  src/blake2sp.c)

target_include_directories(b2 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" src)
target_compile_definitions(b2 PRIVATE SUFFIX=)

install(TARGETS b2 ARCHIVE DESTINATION lib)
install(FILES src/blake2.h DESTINATION include)
